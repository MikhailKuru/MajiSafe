    // Disaster Alerts (ReliefWeb API)

    const ALERT_STORAGE_KEY = 'majiSafe_alerts';


    async function checkRisk() {

      const province = document.getElementById('province').value;

      const district = document.getElementById('district').value.trim();

      const alertBox = document.getElementById('alertBox');


      if (!province || !district) {

        alert('Please enter your province and district');

        return;

      }


      alertBox.style.display = 'block';

      alertBox.className = 'alert warning';

      alertBox.innerHTML = 'Checking live disaster alerts...';


      try {

        const events = await fetchReliefWebEvents(province, district);

        const risk = evaluateRisk(events);


        if (risk.level === 'danger') {

          alertBox.className = 'alert danger';

        } else if (risk.level === 'warning') {

          alertBox.className = 'alert warning';

        } else {

          alertBox.className = 'alert safe';

        }


        alertBox.innerHTML = risk.message;


        // Save for offline mode

        localStorage.setItem(ALERT_STORAGE_KEY, JSON.stringify(events));


        // Push notification for new alerts

        if (risk.level !== 'safe') {

          notifyUser(risk.message);

        }


      } catch (e) {

        alertBox.className = 'alert warning';

        alertBox.innerHTML = 'Unable to fetch live alerts. Showing cached data if available.';


        const cached = localStorage.getItem(ALERT_STORAGE_KEY);

        if (cached) {

          const events = JSON.parse(cached);

          const risk = evaluateRisk(events);

          alertBox.innerHTML = risk.message;

        }

      }

    }


    async function fetchReliefWebEvents(province, district) {

      const url = `https://api.reliefweb.int/v1/reports?appname=majiSafe&filter[field]=country&filter[value]=Mozambique&filter[field]=fields&filter[value]=title&filter[value]=description&limit=20&sort[]=date:desc`;


      const response = await fetch(url);

      const data = await response.json();


      // Filter locally by province/district keywords

      const events = data.data || [];

      const keyword = `${province} ${district}`.toLowerCase();


      return events.filter(e => {

        const title = (e.fields.title || '').toLowerCase();

        const desc = (e.fields.description || '').toLowerCase();

        return title.includes(keyword) || desc.includes(keyword);

      });

    }


    function evaluateRisk(events) {

      if (!events || events.length === 0) {

        return { level: 'safe', message: '✅ SAFE: No active disaster alerts found for your location.' };

      }


      const eventNames = events.slice(0, 3).map(e => e.fields.title);

      const message = `⚠️ ALERT: ${events.length} events found.\n\nTop alerts:\n- ${eventNames.join('\n- ')}`;


      return { level: 'danger', message };

    }


    // -------------------------------

    // Push Notifications

    // -------------------------------

    async function notifyUser(message) {

      if (!("Notification" in window)) return;

      if (Notification.permission === "granted") {

        new Notification("MajiSafe Alert", { body: message });

      } else if (Notification.permission !== "denied") {

        const permission = await Notification.requestPermission();

        if (permission === "granted") {

          new Notification("MajiSafe Alert", { body: message });

        }

      }

    }
